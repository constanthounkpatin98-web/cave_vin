//===============================================
// Filtres : catégorie + recherche
//===============================================

$id_categorie_filtre = isset($_GET["categorie"]) ? $_GET["categorie"] : "";
$recherche = isset($_GET["recherche"]) ? trim($_GET["recherche"]) : "";

$conditions = [
    "vin.statut = 'Disponible'",
    "vin.quantite_stock > 0"
];

$parametres = [];


if (!empty($id_categorie_filtre)) {

    $conditions[] = "vin.id_categorie = ?";
    $parametres[] = $id_categorie_filtre;

}


if (!empty($recherche)) {

    $conditions[] = "vin.nom_vin LIKE ?";
    $parametres[] = "%" . $recherche . "%";

}


$sql_conditions = implode(" AND ", $conditions);


//===============================================
// Récupération des vins
//===============================================

$requete = $connexion->prepare("

    SELECT
        vin.*,
        categorie.libelle AS libelle_categorie

    FROM vin

    LEFT JOIN categorie
        ON vin.id_categorie = categorie.id_categorie

    WHERE $sql_conditions

    ORDER BY vin.nom_vin ASC

");

$requete->execute($parametres);


//===============================================
// Liste des catégories
//===============================================

$requete_categories = $connexion->query("

    SELECT *
    FROM categorie
    WHERE statut = 'Actif'
    ORDER BY libelle ASC

");


//===============================================
// Nombre d'articles dans le panier
//===============================================

$nombre_panier = isset($_SESSION["panier"])
    ? array_sum($_SESSION["panier"])
    : 0;


//===============================================
// Nombre de notifications non lues (seulement si connecté)
//===============================================

$nombre_notifications = 0;
$nombre_commandes_client = 0;

if (isset($_SESSION["client_id"])) {
    $requete_notif = $connexion->prepare("
        SELECT COUNT(*) AS total
        FROM notification
        WHERE id_client = ? AND statut = 'Non lue'
    ");
    $requete_notif->execute([$_SESSION["client_id"]]);
    $nombre_notifications = $requete_notif->fetch()["total"];

    $requete_commandes = $connexion->prepare("
        SELECT COUNT(*) AS total
        FROM commande
        WHERE id_client = ?
    ");
    $requete_commandes->execute([$_SESSION["client_id"]]);
    $nombre_commandes_client = $requete_commandes->fetch()["total"];
}

<script>

    //===========================================================
    // Rotation 3D interactive de la bouteille du hero
    // (léger balancement continu + contrôle à la souris / au doigt,
    //  angle toujours limité pour que la bouteille ne disparaisse jamais)
    //===========================================================

    (function () {

        const bouteille   = document.getElementById("bouteilleHero");
        const conteneur   = document.getElementById("conteneurBouteille");

        if (!bouteille || !conteneur) {
            return;
        }

        const ANGLE_MAX      = 20;   // degrés max de chaque côté : jamais assez pour disparaître
        const VITESSE_AUTO   = 0.04; // vitesse du balancement automatique

        let temps            = 0;
        let angle            = 0;
        let angleManuel      = null;
        let enGlissement     = false;
        let dernierX         = 0;
        let enPause          = false;

        function appliquerRotation() {

            bouteille.style.transform =
                "rotateY(" + angle + "deg)";

        }

        function boucleAnimation() {

            if (!enGlissement && !enPause) {

                if (angleManuel !== null) {

                    // retour en douceur vers le balancement automatique
                    angleManuel += (0 - angleManuel) * 0.05;

                    if (Math.abs(angleManuel) < 0.5) {
                        angleManuel = null;
                    }

                    angle = angleManuel;

                } else {

                    temps += VITESSE_AUTO;
                    angle = Math.sin(temps) * ANGLE_MAX;

                }

                appliquerRotation();

            }

            requestAnimationFrame(boucleAnimation);

        }

        requestAnimationFrame(boucleAnimation);


        function demarrerGlissement(x) {
            enGlissement = true;
            dernierX = x;
        }

        function deplacerGlissement(x) {

            if (!enGlissement) {
                return;
            }

            const delta = x - dernierX;

            angle += delta * 0.4;

            if (angle > ANGLE_MAX)  angle = ANGLE_MAX;
            if (angle < -ANGLE_MAX) angle = -ANGLE_MAX;

            dernierX = x;
            appliquerRotation();

        }

        function terminerGlissement() {

            if (!enGlissement) {
                return;
            }

            enGlissement = false;
            angleManuel = angle;

        }


        // ---- Souris ----

        conteneur.addEventListener("mousedown", function (e) {
            demarrerGlissement(e.clientX);
        });

        window.addEventListener("mousemove", function (e) {
            deplacerGlissement(e.clientX);
        });

        window.addEventListener("mouseup", terminerGlissement);


        // ---- Tactile (mobile / tablette) ----

        conteneur.addEventListener("touchstart", function (e) {
            demarrerGlissement(e.touches[0].clientX);
        }, { passive: true });

        window.addEventListener("touchmove", function (e) {
            deplacerGlissement(e.touches[0].clientX);
        }, { passive: true });

        window.addEventListener("touchend", terminerGlissement);


        // ---- Pause au survol pour mieux regarder l'étiquette ----

        conteneur.addEventListener("mouseenter", function () {
            enPause = true;
        });

        conteneur.addEventListener("mouseleave", function () {
            enPause = false;
        });

    })();

</script>

<script>

/* =====================================================
   ELEMENTS CHATBOT
===================================================== */

const chatbotWindow =
    document.getElementById("chatbotWindow");

const chatbotInput =
    document.getElementById("chatbotInput");

const chatbotMessages =
    document.getElementById("chatbotMessages");

const chatbotSend =
    document.getElementById("chatbotSend");


/* =====================================================
   OUVRIR CHATBOT
===================================================== */

function ouvrirChatbot() {

    chatbotWindow.classList.add("active");

    chatbotInput.focus();

}


/* =====================================================
   FERMER CHATBOT
===================================================== */

function fermerChatbot() {

    chatbotWindow.classList.remove("active");

}


/* =====================================================
   PROTECTION TEXTE
===================================================== */

function securiserTexte(texte) {

    const element =
        document.createElement("div");

    element.textContent = texte;

    return element.innerHTML;

}


/* =====================================================
   MESSAGE CLIENT
===================================================== */

function ajouterMessageClient(texte) {

    const message =
        document.createElement("div");

    message.className =
        "chatbot-message chatbot-client";


    message.innerHTML = `

        <div class="chatbot-bubble">

            ${securiserTexte(texte)}

        </div>

    `;


    chatbotMessages.appendChild(message);


    chatbotMessages.scrollTop =
        chatbotMessages.scrollHeight;

}


/* =====================================================
   MESSAGE BOT
===================================================== */

function ajouterMessageBot(texte) {

    const message =
        document.createElement("div");

    message.className =
        "chatbot-message chatbot-bot";


    message.innerHTML = `

        <div class="chatbot-mini-avatar">

            🍷

        </div>


        <div class="chatbot-bubble">

            ${texte}

        </div>

    `;


    chatbotMessages.appendChild(message);


    chatbotMessages.scrollTop =
        chatbotMessages.scrollHeight;

}


/* =====================================================
   ENVOYER MESSAGE
===================================================== */

async function envoyerMessageChatbot() {

    const texte =
        chatbotInput.value.trim();


    if (texte === "") {

        return;

    }


    /* Message client */

    ajouterMessageClient(texte);


    chatbotInput.value = "";

    chatbotSend.disabled = true;


    /* Chargement */

    const chargement =
        document.createElement("div");


    chargement.id =
        "chatbotChargement";


    chargement.className =
        "chatbot-message chatbot-bot";


    chargement.innerHTML = `

        <div class="chatbot-mini-avatar">

            🍷

        </div>


        <div class="chatbot-bubble">

            <span class="spinner-border spinner-border-sm me-2"></span>

            Je réfléchis...

        </div>

    `;


    chatbotMessages.appendChild(chargement);


    chatbotMessages.scrollTop =
        chatbotMessages.scrollHeight;


    try {


        const donnees =
            new FormData();


        donnees.append(
            "message",
            texte
        );


        const reponse =
            await fetch(
                "chatbot_api.php",
                {
                    method: "POST",
                    body: donnees
                }
            );


        const resultat =
            await reponse.json();


        /* Supprimer chargement */

        const chargementActuel =
            document.getElementById(
                "chatbotChargement"
            );


        if (chargementActuel) {

            chargementActuel.remove();

        }


        if (resultat.success) {

            ajouterMessageBot(
                resultat.response
            );

        } else {

            ajouterMessageBot(

                "⚠️ " +

                (
                    resultat.message ||
                    "Une erreur est survenue."
                )

            );

        }

    }

    catch (erreur) {

        console.error(erreur);


        const chargementActuel =
            document.getElementById(
                "chatbotChargement"
            );


        if (chargementActuel) {

            chargementActuel.remove();

        }


        ajouterMessageBot(

            "❌ Impossible de contacter le serveur."

        );

    }


    chatbotSend.disabled = false;

    chatbotInput.focus();

}
